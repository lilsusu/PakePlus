1.4 关键约束

  1. Python 不直接操作 MySQL，所有数据读写通过 Laravel API。
  2. Laravel 不阻塞等待 Python，任务全程异步。
  3. 积分扣费由 Laravel 统一管理：预扣 → 结算 → 失败退回。
  4. 所有 AI 任务进入队列，支持进度推送、失败重试、超时取消。
  5. 所有 Key 加密存储，后台脱敏显示，日志不得记录完整 Key。
  6. 所有写操作幂等，支付回调验签，任务可重试。
  7. 所有关键操作记录审计日志。

步骤说明:
1. 前端 POST /api/ai/tasks → Laravel 创建 AiTask 记录
2. Laravel 预扣积分(冻结) → 推入 Redis 队列
3. Celery Worker 拉取任务 → 调用 Laravel API 获取加密 Key(脱敏后传递)
4. Worker 调用外部 AI 平台 API
5. Worker POST /api/webhook/ai-callback → Laravel 更新状态
6. Laravel 结算积分(成功)/退回积分(失败)
7. Laravel 通过 Reverb 推送进度到前端

3数据库 ER 设计

数据库分为 5 个领域：基础与认证、支付与积分、AI 平台与任务、提示词工程、工作台与画布。以下按领域给出建表 SQL。

3.1 基础与认证表

-- ============================================================
-- 1. 用户表
-- ============================================================
CREATE TABLE users (
    id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name            VARCHAR(100) NOT NULL,
    email           VARCHAR(255) NULL,
    phone           VARCHAR(20) NULL,
    phone_verified_at DATETIME NULL,
    email_verified_at DATETIME NULL,
    password        VARCHAR(255) NOT NULL,
    avatar          VARCHAR(500) NULL,
    real_name_status ENUM('none','pending','approved','rejected') DEFAULT 'none',
    real_name_data  JSON NULL,           -- 实名认证信息(加密)
    status          ENUM('active','banned') DEFAULT 'active',
    ban_reason      VARCHAR(500) NULL,
    ban_expires_at  DATETIME NULL,
    member_level    TINYINT DEFAULT 0,   -- 会员等级
    member_expires_at DATETIME NULL,
    invite_code     VARCHAR(32) UNIQUE,  -- 邀请码
    inviter_id      BIGINT UNSIGNED NULL,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at      DATETIME NULL,
    INDEX idx_phone (phone),
    INDEX idx_email (email),
    INDEX idx_inviter (inviter_id),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 2. 角色表 (spatie/laravel-permission 兼容)
-- ============================================================
CREATE TABLE roles (
    id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name            VARCHAR(125) NOT NULL,
    guard_name      VARCHAR(125) NOT NULL DEFAULT 'web',
    display_name    VARCHAR(100) NOT NULL,
    description     VARCHAR(500) NULL,
    data_scope      ENUM('all','self','department') DEFAULT 'self',
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_name_guard (name, guard_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 3. 权限表
-- ============================================================
CREATE TABLE permissions (
    id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name            VARCHAR(125) NOT NULL,  -- 如 users.read, ai-keys.write
    guard_name      VARCHAR(125) NOT NULL DEFAULT 'web',
    display_name    VARCHAR(100) NOT NULL,
    module          VARCHAR(50) NOT NULL,    -- 所属模块
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY uk_name_guard (name, guard_name),
    INDEX idx_module (module)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 4. 角色-权限关联表
-- ============================================================
CREATE TABLE role_has_permissions (
    permission_id   BIGINT UNSIGNED NOT NULL,
    role_id         BIGINT UNSIGNED NOT NULL,
    PRIMARY KEY (permission_id, role_id),
    FOREIGN KEY (permission_id) REFERENCES permissions(id) ON DELETE CASCADE,
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 5. 用户-角色关联表
-- ============================================================
CREATE TABLE model_has_roles (
    role_id         BIGINT UNSIGNED NOT NULL,
    model_type      VARCHAR(125) NOT NULL DEFAULT 'App\\Models\\User',
    model_id        BIGINT UNSIGNED NOT NULL,
    PRIMARY KEY (role_id, model_id, model_type),
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE,
    INDEX idx_model (model_id, model_type)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 6. 审计日志表
-- ============================================================
CREATE TABLE audit_logs (
    id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id         BIGINT UNSIGNED NULL,
    user_type       VARCHAR(125) NULL,
    event           VARCHAR(50) NOT NULL,     -- created/updated/deleted/custom
    auditable_type  VARCHAR(125) NOT NULL,
    auditable_id    BIGINT UNSIGNED NOT NULL,
    old_values       JSON NULL,
    new_values       JSON NULL,
    url              VARCHAR(500) NULL,
    ip_address      VARCHAR(45) NULL,
    user_agent      VARCHAR(500) NULL,
    tags             VARCHAR(255) NULL,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_auditable (auditable_type, auditable_id),
    INDEX idx_user (user_id),
    INDEX idx_event (event),
    INDEX idx_created (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 7. 系统通知表
-- ============================================================
CREATE TABLE notifications (
    id              CHAR(36) PRIMARY KEY,
    type            VARCHAR(125) NOT NULL,
    notifiable_type VARCHAR(125) NOT NULL,
    notifiable_id   BIGINT UNSIGNED NOT NULL,
    data            JSON NOT NULL,
    read_at         DATETIME NULL,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_notifiable (notifiable_type, notifiable_id),
    INDEX idx_read (read_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

3.2 支付与积分表

-- ============================================================
-- 8. 用户积分钱包表
-- ============================================================
CREATE TABLE wallets (
    id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id         BIGINT UNSIGNED NOT NULL UNIQUE,
    balance         BIGINT DEFAULT 0 NOT NULL,    -- 可用积分
    frozen          BIGINT DEFAULT 0 NOT NULL,    -- 冻结积分(预扣)
    total_recharged BIGINT DEFAULT 0 NOT NULL,    -- 累计充值
    total_consumed  BIGINT DEFAULT 0 NOT NULL,    -- 累计消费
    total_refunded  BIGINT DEFAULT 0 NOT NULL,    -- 累计退回
    expires_at      DATETIME NULL,                 -- 积分过期时间
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_balance (balance)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 9. 积分流水表
-- ============================================================
CREATE TABLE points_ledger (
    id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    wallet_id       BIGINT UNSIGNED NOT NULL,
    user_id         BIGINT UNSIGNED NOT NULL,
    type            ENUM('income','expense','frozen','unfrozen','refund','adjust') NOT NULL,
    amount          BIGINT NOT NULL,               -- 正数增加/负数减少
    balance_after   BIGINT NOT NULL,               -- 变动后余额
    related_type    VARCHAR(125) NULL,             -- 关联对象类型(Order/AiTask)
    related_id      BIGINT UNSIGNED NULL,           -- 关联对象 ID
    reason          VARCHAR(500) NULL,
    operator_id     BIGINT UNSIGNED NULL,           -- 操作人(手动调整时)
    idempotency_key VARCHAR(128) UNIQUE,            -- 幂等键
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (wallet_id) REFERENCES wallets(id) ON DELETE CASCADE,
    INDEX idx_user (user_id),
    INDEX idx_type (type),
    INDEX idx_related (related_type, related_id),
    INDEX idx_created (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ============================================================
-- 10. 充值套餐表
-- ============================================================
CREATE TABLE recharge_packages (
    id              BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name            VARCHAR(100) NOT NULL,
    description    VARCHAR(500) NULL,
    price_cents     INT UNSIGNED NOT NULL,         -- 价格(分)
    base_points     BIGINT NOT NULL,                -- 基础积分

    member_level    TINYINT DEFAULT 0,               -- 关联会员等级

    sort_order      INT DEFAULT 0,

    starts_at       DATETIME NULL,
    ends_at         DATETIME NULL,
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

-- ============================================================

class BaseAdapter(ABC):
    """所有 AI 平台适配器的基类"""

    @abstractmethod
    async def generate(self, prompt: str, variables: dict, params: dict) -> dict:
        """生成内容(文生文/文生图/文生视频)
        Returns: {result, cost_cents, latency_ms, metadata}
        """
        pass

    @abstractmethod
    async def edit(self, input_data: bytes, prompt: str, params: dict) -> dict:
        """编辑内容(图生图/inpaint/图生视频)"""
        pass

    @abstractmethod
    async def compose(self, inputs: list, params: dict) -> dict:
        """合成内容(视频拼接/FFmpeg)"""
        pass
